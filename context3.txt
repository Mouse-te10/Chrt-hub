from sqlalchemy import Column, Integer, String, Text, Boolean, ForeignKey, UniqueConstraint, ARRAY
from sqlalchemy.orm import declarative_base, relationship

Base = declarative_base()

class User(Base):
    __tablename__ = "users"

    id = Column(Integer, primary_key=True, index=True)
    username = Column(String(50), unique=True, nullable=False, index=True)
    hashed_password = Column(String(255), nullable=False)
    group_name = Column(String(20), nullable=False)  # Чистое название группы (например, Ис-381)
    about = Column(Text, nullable=True)

    # Связи
    projects = relationship("Project", back_populates="author", cascade="all, delete-orphan")
    comments = relationship("Comment", back_populates="author", cascade="all, delete-orphan")
    votes = relationship("CommentVote", back_populates="user", cascade="all, delete-orphan")


class Project(Base):
    __tablename__ = "projects"

    id = Column(Integer, primary_key=True, index=True)
    title = Column(String(100), nullable=False, index=True)
    short_desc = Column(String(255), nullable=False)
    full_desc = Column(Text, nullable=True)
    direction = Column(String(50), nullable=False)  # Веб-разработка, Дизайн, Game-dev и т.д.
    lang = Column(String(30), nullable=True)         # Python, JS, C++
    github = Column(String(255), nullable=True)
    comments_disabled = Column(Boolean, default=False)
    author_id = Column(Integer, ForeignKey("users.id", ondelete="CASCADE"), nullable=False)

    # Связи
    author = relationship("User", back_populates="projects")
    comments = relationship("Comment", back_populates="project", cascade="all, delete-orphan")
    media_urls = Column(ARRAY(String), nullable=True, default=[])
    likes = relationship("ProjectLike", cascade="all, delete-orphan")


class Comment(Base):
    __tablename__ = "comments"

    id = Column(Integer, primary_key=True, index=True)
    text = Column(Text, nullable=False)
    author_id = Column(Integer, ForeignKey("users.id", ondelete="CASCADE"), nullable=False)
    project_id = Column(Integer, ForeignKey("projects.id", ondelete="CASCADE"), nullable=False)

    # Связи
    author = relationship("User", back_populates="comments")
    project = relationship("Project", back_populates="comments")
    votes = relationship("CommentVote", back_populates="comment", cascade="all, delete-orphan")


class CommentVote(Base):
    """Таблица для хранения лайков и дизлайков (один юзер - один голос)"""
    __tablename__ = "comment_votes"

    id = Column(Integer, primary_key=True, index=True)
    user_id = Column(Integer, ForeignKey("users.id", ondelete="CASCADE"), nullable=False)
    comment_id = Column(Integer, ForeignKey("comments.id", ondelete="CASCADE"), nullable=False)
    is_like = Column(Boolean, nullable=False)  # True = Лайк, False = Дизлайк

    # Связи
    user = relationship("User", back_populates="votes")
    comment = relationship("Comment", back_populates="votes")

    # Жесткое ограничение: один пользователь может проголосовать за один коммент только один раз
    __table_args__ = (UniqueConstraint('user_id', 'comment_id', name='_user_comment_vote_uc'),)


class ProjectLike(Base):
    """Таблица для хранения лайков на сами проекты (один юзер - один лайк на проект)"""

from datetime import datetime, timedelta, timezone
from typing import Optional
from jose import jwt
import os
import bcrypt

# Безопасно читаем ключ из .env. Если его нет — падаем с ошибкой при запуске
SECRET_KEY = os.getenv("JWT_SECRET_KEY")
if not SECRET_KEY:
    raise RuntimeError(
        "🚨 КРИТИЧЕСКАЯ ОШИБКА БЕЗОПАСНОСТИ: Переменная окружения JWT_SECRET_KEY не задана в .env или Docker Compose!"
    )

ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 1440

def hash_password(password: str) -> str:
    pwd_bytes = password.encode('utf-8')
    salt = bcrypt.gensalt()
    hashed = bcrypt.hashpw(pwd_bytes, salt)
    return hashed.decode('utf-8')

def verify_password(plain_password: str, hashed_password: str) -> bool:
    return bcrypt.checkpw(
        plain_password.encode('utf-8'),
        hashed_password.encode('utf-8')
    )

def create_access_token(data: dict, expires_delta: timedelta | None = None):
    to_encode = data.copy()
    if expires_delta:
        expire = datetime.now(timezone.utc) + expires_delta
    else:
        expire = datetime.now(timezone.utc) + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

@app.post("/api/auth/login", response_model=Token, tags=["Auth"])
async def login_user(login_data: UserLogin, db: AsyncSession = Depends(get_db)):
    # Ищем пользователя по логину
    result = await db.execute(select(User).where(User.username == login_data.username))
    user = result.scalars().first()
    
    if not user or not verify_password(login_data.password, user.hashed_password):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Неверное имя пользователя или пароль!"
        )
    
    # Считаем, сколько проектов у этого пользователя уже есть в базе данных
    count_result = await db.execute(select(func.count(Project.id)).where(Project.author_id == user.id))
    project_count = count_result.scalar() or 0
    
    # Динамически добавляем это свойство в объект user перед отправкой, чтобы Pydantic-схема его сожрала
    user.project_count = project_count
    
    access_token = create_access_token(data={"sub": user.username, "user_id": user.id})
    return {
        "access_token": access_token,
        "token_type": "bearer",
        "user": user
    }

# Переписываем зависимость для извлечения текущего пользователя
async def get_current_user(
    request: Request, # Добавляем request для чтения заголовков вручную
    db: AsyncSession = Depends(get_db)
):
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Не удалось валидировать учетные данные",
        headers={"WWW-Authenticate": "Bearer"},
    )
    
    token = None
    
    # 1. Сначала пробуем достать токен из стандартного заголовка Authorization
    auth_header = request.headers.get("Authorization")
    if auth_header and auth_header.startswith("Bearer "):
        token = auth_header.split(" ")[1]
        
    # 2. Если заголовка нет (особенность multipart/form-data на фронте), 
    # пробуем прочитать его из полей формы (Form Data)
    if not token:
        try:
            form_data = await request.form()
            token = form_data.get("token") # Будем передавать его с фронта как обычное поле
        except:
            pass

    if not token:
        raise credentials_exception

    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")
        user_id: int = payload.get("user_id")
        if username is None or user_id is None:
            raise credentials_exception
    except JWTError:
        raise credentials_exception
        
    result = await db.execute(select(User).where(User.id == user_id))
    user = result.scalars().first()
    if user is None:
        raise credentials_exception
    return user

@app.get("/api/projects", tags=["Projects"])
async def get_all_projects(direction: str = "Все", lang: str = "Все", request: Request = None, db: AsyncSession = Depends(get_db)):
    # Безопасно извлекаем user_id из JWT-токена
    user_id = None
    auth_header = request.headers.get("Authorization") if request else None
    if auth_header and auth_header.startswith("Bearer "):
        try:
            token = auth_header.split(" ")[1]
            payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
            user_id = payload.get("user_id")
        except:
            pass

    # 1. Создаем изолированный подзапрос для подсчета лайков (избавляет от гигантских group_by)
    likes_subquery = (
        select(
            ProjectLike.project_id,
            func.count(ProjectLike.id).label("likes_count")
        )
        .group_by(ProjectLike.project_id)
        .subquery()
    )

    # 2. Основной запрос: соединяем проект, автора и наш подзапрос лайков
    query = (
        select(
            Project,
            func.coalesce(likes_subquery.c.likes_count, 0).label("likes_count")
        )
        .options(selectinload(Project.author))
        .join(likes_subquery, Project.id == likes_subquery.c.project_id, isouter=True)
    )
    
    # Фильтрация по направлению
    if direction != "Все":
        query = query.where(Project.direction == direction)
        
    # Безопасная фильтрация по языку программирования (с защитой от None / null в БД)
    if lang != "Все":
        query = query.where(
            Project.lang.isnot(None),
            func.lower(Project.lang) == lang.lower()
        )
    
    result = await db.execute(query.order_by(Project.id.desc()))
    rows = result.all()
    
    response_data = []
    for row in rows:
        p = row[0]          # Объект Project
        likes_cnt = row[1]  # Гарантированное число лайков из подзапроса
        
        # Проверяем, лайкнул ли проект именно этот залогиненный студент
        has_liked = False
        if user_id:
            like_check = await db.execute(
                select(ProjectLike).where(ProjectLike.user_id == user_id, ProjectLike.project_id == p.id)
            )
            has_liked = like_check.scalars().first() is not None

        response_data.append({
            "id": p.id,
            "title": p.title,
            "shortDesc": p.short_desc,
            "fullDesc": p.full_desc,
            "direction": p.direction,
            "lang": p.lang,
            "github": p.github,
            "commentsDisabled": p.comments_disabled,
            "author": p.author.username,
            "author_id": p.author_id,
            "group_name": p.author.group_name,
            "media": p.media_urls,
            "likesCount": int(likes_cnt),  # Приводим к чистому Int
            "hasLiked": has_liked
        })
        
    return response_data

@app.get("/api/projects/{project_id}/comments", response_model=list[CommentResponse], tags=["Comments"])
async def get_project_comments(project_id: int, request: Request, db: AsyncSession = Depends(get_db)):
    # 1. Вручную и безопасно проверяем наличие заголовка Authorization
    user_id = None
    auth_header = request.headers.get("Authorization")
    
    if auth_header and auth_header.startswith("Bearer "):
        try:
            token = auth_header.split(" ")[1]
            payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
            user_id = payload.get("user_id")
        except JWTError:
            pass  # Если токен сломан или истек, просто читаем как аноним

    # 2. Запрашиваем комментарии из базы (этот код выполняется теперь и для анонимов, и для студентов!)
    result = await db.execute(
        select(Comment)
        .options(selectinload(Comment.author), selectinload(Comment.votes))
        .where(Comment.project_id == project_id)
        .order_by(Comment.id.asc())
    )
    comments = result.scalars().all()

    response = []
    for c in comments:
        likes = sum(1 for v in c.votes if v.is_like)
        dislikes = sum(1 for v in c.votes if not v.is_like)
        
        # Выясняем, как проголосовал текущий пользователь (если он авторизован)
        user_vote = None
        if user_id:
            for v in c.votes:
                if v.user_id == user_id:
                    user_vote = v.is_like
                    break

        response.append({
            "id": c.id,
            "text": c.text,
            "project_id": c.project_id,
            "author_id": c.author_id,
            "author_name": c.author.username,
            "author_group": c.author.group_name,
            "likes_count": likes,
            "dislikes_count": dislikes,
            "user_vote": user_vote
        })
    
    return response

@app.post("/api/comments/{comment_id}/vote", tags=["Comments"])
async def vote_comment(comment_id: int, vote_data: VoteRequest, current_user: User = Depends(get_current_user), db: AsyncSession = Depends(get_db)):
    # Проверяем наличие комментария
    comm_res = await db.execute(select(Comment).where(Comment.id == comment_id))
    if not comm_res.scalars().first():
        raise HTTPException(status_code=404, detail="Комментарий не найден")

    # ИСПРАВЛЕНО: Теперь ищем голос ИМЕННО ЭТОГО залогиненного студента (защита от IDOR)
    vote_res = await db.execute(
        select(CommentVote)
        .where(CommentVote.user_id == current_user.id, CommentVote.comment_id == comment_id)
    )
    existing_vote = vote_res.scalars().first()

    if existing_vote:
        if existing_vote.is_like == vote_data.is_like:
            # Если кликнули на то же самое состояние — забираем голос (Toggle OFF)
            await db.delete(existing_vote)
            msg = "Голос убран"
        else:
            # Если кликнули на противоположное состояние — меняем лайк на дизлайк (или наоборот)
            existing_vote.is_like = vote_data.is_like
            msg = "Голос изменен"
    else:
        # Голоса не было — создаем новую запись
        new_vote = CommentVote(
            user_id=current_user.id,
            comment_id=comment_id,
            is_like=vote_data.is_like
        )
        db.add(new_vote)
        msg = "Голос учтен"

    await db.commit()
    return {"status": "success", "message": msg}


@app.get("/api/users/rating", tags=["System"])
async def get_users_rating(db: AsyncSession = Depends(get_db)):

    query = (
        select(
            User.username, 
            User.group_name, 
            func.count(distinct(ProjectLike.id)).label("total_likes")
        )
        .join(Project, User.id == Project.author_id, isouter=True)
        .join(ProjectLike, Project.id == ProjectLike.project_id, isouter=True)
        .group_by(User.id, User.username, User.group_name)
        .order_by(func.count(distinct(ProjectLike.id)).desc(), User.username.asc())
        .limit(5)
    )
    
    result = await db.execute(query)
    rating_data = result.all()
    
    return [
        {
            "name": f"{r.username} ({r.group_name})",
            "count": r.total_likes
        }
        for r in rating_data
    ]

  const [shortDesc, setShortDesc] = useState('');
  const [fullDesc, setFullDesc] = useState('');
  const [direction, setDirection] = useState('Веб-разработка');
  const [lang, setLang] = useState('Python');
  const [github, setGithub] = useState('');
  const [disableComments, setDisableComments] = useState(false);
  
  // Состояние для хранения выбранных медиафайлов
  const [mediaFiles, setMediaFiles] = useState([]);
  const fileInputRef = useRef(null);

  // === ДОБАВЛЕННАЯ ФУНКЦИЯ ДЛЯ ОБНОВЛЕНИЯ СТЕЙТА ===
  const addFileToState = (file) => {
    const fileType = file.type.startsWith('video/') ? 'video' : 'image';
    const newFileObject = {
      raw: file, // Храним оригинальный файл для отправки на бэк
      type: fileType,
      preview: URL.createObjectURL(file) // Создаем временную ссылку для тегов <img> / <video>
    };
    setMediaFiles(prev => [...prev, newFileObject]);
  };

  const handleFileChange = (e) => {
    const files = Array.from(e.target.files);
    
    // Проверка 1: Лимит на общее количество (до 10 штук суммарно)
    if (mediaFiles.length + files.length > 10) {
      alert('Максимальное количество медиафайлов (фото + видео) — 10 штук!');
      return;
    }

    files.forEach(file => {
      // Проверка 2: Если это видео, проверяем его длину
      if (file.type.startsWith('video/')) {
        const videoElement = document.createElement('video');
        videoElement.preload = 'metadata';
        
        videoElement.onloadedmetadata = () => {
          window.URL.revokeObjectURL(videoElement.src);
          // Длина видео не должна превышать 30 секунд
          if (videoElement.duration > 30) {
            alert(`Видео "${file.name}" превышает лимит в 30 секунд! Загрузка отменена.`);
          } else {
            addFileToState(file);
          }
        };
        videoElement.src = URL.createObjectURL(file);
      } else if (file.type.startsWith('image/')) {
        // Если это фото, добавляем сразу
        addFileToState(file);
      } else {
        alert('Разрешено загружать только фотографии и видеофайлы!');
      }
    });

    // Сбрасываем значение инпута, чтобы можно было выбрать тот же файл повторно
    if (fileInputRef.current) fileInputRef.current.value = '';
  };

  const removeFile = (index) => {
    const fileToRemove = mediaFiles[index];
    if (fileToRemove && fileToRemove.preview) {
      URL.revokeObjectURL(fileToRemove.preview);
    }
    setMediaFiles(prev => prev.filter((_, i) => i !== index));
  };

  const handleFormSubmit = (e) => {

    if (!title || !shortDesc) {

    });

    setTitle('');
    setShortDesc('');
    setFullDesc('');
    setGithub('');

  };

function SidebarRating({ refreshTrigger }) {
  const [leaders, setLeaders] = useState([]);
  const [loading, setLoading] = useState(true);

  // Подгружаем рейтинг при монтировании и при триггере (например, создании/удалении проекта)
  useEffect(() => {
    const loadRating = async () => {
      try {
        const data = await projectService.getLeaderboard();
        setLeaders(data);
      } catch (err) {
        console.error('Ошибка рейтинга:', err.message);
      } finally {
        setLoading(false);
      }
    };
    loadRating();
  }, [refreshTrigger]);

  return (
    <aside className="rating-sidebar">
      <h3 className="rating-title">Рейтинг</h3>
      <div className="rating-list">
        {loading ? (
          <div style={{ color: 'var(--text-muted)', fontSize: '0.9rem' }}>Загрузка топа...</div>
        ) : leaders.length > 0 ? (
          leaders.map((student, i) => (
            <div key={i} className="rating-item">
              <span className="user-name">{student.name}</span>
              <span className="user-score">{student.count}</span>
            </div>
          ))
        ) : (
          <div style={{ color: 'var(--text-muted)', fontSize: '0.9rem' }}>Пусто</div>

import { useState } from 'react';
import './LoginPage.css';

function LoginPage({ onBack, onSubmitSuccess }) {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [isLoading, setIsLoading] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!username || !password) {
      alert('Заполните все поля!');
      return;
    }

    setIsLoading(true);
    try {
      const response = await fetch('/api/auth/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username, password })
      });

      const data = await response.json();

      if (!response.ok) {
        // Если бэкенд вернул ошибку (401 и т.д.), выводим её текст
        throw new Error(data.detail || 'Ошибка при входе');
      }

      // Сохраняем JWT-токен в localStorage, чтобы сессия не сбрасывалась
      localStorage.setItem('token', data.access_token);
      localStorage.setItem('user', JSON.stringify(data.user));

      // Передаем данные наверх в App.jsx
      onSubmitSuccess(data.user);
    } catch (error) {
      alert(error.message);
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="login-page-wrapper">
      <div className="bg-glow glow-top-left"></div>
      <div className="bg-glow glow-bottom-right"></div>

      <div className="login-page-container">
        <h1 className="auth-title">Chrt-hub</h1>

        <form className="register-form" onSubmit={handleSubmit}>
          <div className="form-group">
            <label>Имя пользователя</label>
            <input 
              type="text" 
              placeholder="Введите логин" 
              value={username}
              onChange={(e) => setUsername(e.target.value)}
              disabled={isLoading}
            />
          </div>

          <div className="form-group">
            <label>Пароль</label>
            <input 
              type="password" 
              placeholder="Введите ваш пароль" 
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              disabled={isLoading}
            />
          </div>

          <button type="submit" className="submit-btn" disabled={isLoading}>

import { useState, useEffect } from 'react';
import './ProjectPage.css';
import Header from '../MainPage/Header/Header';
import Footer from '../MainPage/Footer/Footer';
import { projectService } from '../../services/projectService'; // Подключаем сервис

function ProjectPage({ project, currentUsername, onBack, projects, onViewProjectDetails, onOpenMyProjects, user }) {
  const [mediaIndex, setMediaIndex] = useState(0);
  const [isLightboxOpen, setIsLightboxOpen] = useState(false);
  const [pageSearchValue, setPageSearchValue] = useState('');

  // Заменяем хардкод на динамический стейт из БД
  const [comments, setComments] = useState([]);
  const [newCommentText, setNewCommentText] = useState('');
  const [commentsLoading, setCommentsLoading] = useState(true);

  const projectMedia = project?.media && project.media.length > 0 
    ? project.media 
    : ["https://unsplash.com"];

  // Подгружаем комментарии при монтировании страницы или смене ID проекта
  useEffect(() => {
    if (project?.id) {
      loadComments();
    }
  }, [project?.id]);

  const loadComments = async () => {
    try {
      setCommentsLoading(true);
      const data = await projectService.getComments(project.id);
      setComments(data);
    } catch (err) {
      console.error(err.message);
    } finally {
      setCommentsLoading(false);
    }
  };

  const nextMedia = () => setMediaIndex((prev) => (prev === projectMedia.length - 1 ? 0 : prev + 1));
  const prevMedia = () => setMediaIndex((prev) => (prev === 0 ? projectMedia.length - 1 : prev - 1));

  // Реальное голосование через API с оптимистичным обновлением интерфейса
  const handleVote = async (commentId, voteType) => {
    if (user.role === 'anonymous') {
      alert('Голосовать могут только авторизованные студенты ЧРТ!');
      return;
    }

    const isLikeTarget = voteType === 'like';
    
    try {
      // Отправляем запрос на бэк
      await projectService.voteComment(commentId, isLikeTarget);
      // Просто пересобираем комменты из БД, чтобы получить идеальные точные счетчики
      const updatedComments = await projectService.getComments(project.id);
      setComments(updatedComments);
    } catch (err) {
      alert(err.message);
    }
  };

  // Добавление комментария через API
  const handleAddComment = async (e) => {
    e.preventDefault();
    if (user.role === 'anonymous') {
      alert('Анонимы не могут оставлять комментарии. Пожалуйста, войдите в аккаунт!');
      return;
    }
    if (!newCommentText.trim()) return;

    try {
      const createdComment = await projectService.createComment(newCommentText, project.id);
      // Добавляем новый коммент в список локально
      setComments([...comments, createdComment]);
      setNewCommentText('');
    } catch (err) {
      alert(err.message);
    }
  };

  const isDesign = project?.direction === 'Дизайн';

  const authorTotalLikes = projects
    .filter(p => p.author === project?.author)
    .reduce((sum, p) => sum + (p.likesCount || 0), 0);

  return (
    <div className="project-page-root" style={{backgroundColor: 'var(--bg-main)', minHeight: '100vh'}}>
      <div className="bg-glow glow-top-left"></div>
      <div className="bg-glow glow-bottom-right"></div>
      <Header 
        user={user} 
        searchValue={pageSearchValue} 
        onSearchChange={setPageSearchValue} 
        setScreen={(targetScreen) => {
          if (targetScreen === 'my_projects') {
            onOpenMyProjects();
          } else {
            onBack();
          }
        }}
        projects={projects} 
        onViewProjectDetails={onViewProjectDetails} 
        showDropdown={true}
        ViewButtonMyProjects={true}
      />

      <div className="project-sub-header">
        <button className="back-to-hub-btn" onClick={onBack}>
          ← Назад в ленту
        </button>
        {isDesign ? (
          <button className="github-top-btn disabled" disabled>Код недоступен (Дизайн)</button>
        ) : (
          <a href={project?.github || "https://github.com"} target="_blank" rel="noreferrer" className="github-top-btn">
            Перейти на GitHub
          </a>
        )}
      </div>

      <div className="project-details-grid">
        <main className="project-content-left">
          <h2 className="project-main-title">{project?.title || "Название проекта"}</h2>
          <p className="project-long-desc">{project?.fullDesc || project?.shortDesc || "Описание проекта..."}</p>

          <div className="gallery-slider-container">
            <button className="slider-arrow left" onClick={prevMedia}>‹</button>
            <div className="slider-media-window" onClick={() => setIsLightboxOpen(true)}>
              {projectMedia[mediaIndex].includes('video') || (typeof projectMedia[mediaIndex] === 'string' && projectMedia[mediaIndex].endsWith('.mp4')) ? (
                <video src={projectMedia[mediaIndex]} muted autoPlay loop />
              ) : (
                <img src={projectMedia[mediaIndex]} alt="Скриншот проекта" />
              )}
            </div>
            <button className="slider-arrow right" onClick={nextMedia}>›</button>
          </div>

          {/* Блок Комментариев */}
          <section className="project-comments-wrapper">
            <h3 className="comments-block-title">Комментарии ({comments.length})</h3>
            
            {project?.commentsDisabled ? (
              <div style={{color: 'var(--text-muted)', fontStyle: 'italic', padding: '10px 0'}}>
                Автор отключил комментарии к этому проекту.
              </div>
            ) : commentsLoading ? (
              <div style={{color: 'var(--text-muted)'}}>Загрузка обсуждения...</div>
            ) : (
              <>
                <div className="comments-flow-list">
                  {comments.map((c) => (
                    <div key={c.id} className="comment-node-box">
                      <div className="comment-meta-row">
                        <span>
                          {c.author_name} ({c.author_group}) {c.author_name === currentUsername && <span style={{color: 'var(--accent-mint)'}}>(Вы)</span>}
                        </span>
                      </div>
                      <p className="comment-text-body">{c.text}</p>
                      
                      <div className="comment-actions-bar">
                        <div className="voting-buttons-pack">
                          <button 
                            type="button"
                            className={`vote-action-btn liked ${c.user_vote === true ? 'active' : ''}`}
                            onClick={() => handleVote(c.id, 'like')}
                          >
                            ▲ {c.likes_count}
                          </button>
                          <button 
                            type="button"
                            className={`vote-action-btn disliked ${c.user_vote === false ? 'active' : ''}`}
                            onClick={() => handleVote(c.id, 'dislike')}
                          >
                            ▼ {c.dislikes_count}
                          </button>
                        </div>
                      </div>
                    </div>
                  ))}
                  {comments.length === 0 && (
                    <div style={{color: 'var(--text-muted)', padding: '10px 0'}}>Будьте первым, кто оставит отзыв!</div>
                  )}
                </div>

                {user.role !== 'anonymous' && (
                  <form className="add-comment-form-block" onSubmit={handleAddComment}>
                    <input 
                      type="text" 
                      placeholder="Напишите ваш отзыв или вопрос по коду..." 
                      value={newCommentText}
                      onChange={e => setNewCommentText(e.target.value)}
                      maxLength={1000}
                    />
                    <button type="submit">Отправить</button>
                  </form>
                )}
              </>
            )}
          </section>
        </main>

        <aside className="project-author-sidebar">
          <div className="sidebar-heading">Автор проекта</div>
          <div className="author-main-profile">
            <div className="author-huge-name">{project?.author || "Студент ЧРТ"}</div>
            <div className="author-group-tag">Группа: {project?.group_name || "ЧРТ"}</div>
          </div>
          <div className="author-rating-score-box">
            <span style={{ color: 'var(--text-muted)' }}>Рейтинг на хабе:</span>
            {/* Выводим реальный, динамический рейтинг вместо ★ 100 */}
            <span style={{ color: 'var(--accent-mint)', fontWeight: 700 }}>❤️ {authorTotalLikes}</span>
          </div>
          <div className="author-bio-text">
            {project?.authorAbout || "Студент Челябинского Радиотехнического Техникума. Вносит вклад в репозитории комьюнити Chrt-Hub."}
          </div>
        </aside>
      </div>

      {isLightboxOpen && (
        <div className="lightbox-modal-overlay" onClick={() => setIsLightboxOpen(false)}>
          <div className="lightbox-content-box" onClick={e => e.stopPropagation()}>
            <button type="button" className="lightbox-close-top-btn" onClick={() => setIsLightboxOpen(false)}>
              ← Назад
            </button>
            {projectMedia[mediaIndex].includes('video') || (typeof projectMedia[mediaIndex] === 'string' && projectMedia[mediaIndex].endsWith('.mp4')) ? (
              <video src={projectMedia[mediaIndex]} controls autoPlay />
            ) : (
              <img src={projectMedia[mediaIndex]} alt="Увеличенное превью" />
            )}
          </div>
        </div>
      )}
      <Footer />
    </div>
  );
}

export default ProjectPage;

function MyProjectsPage({ user, projects, onBack, onDeleteProject, onViewProjectDetails, onToggleLike }) {
  const [searchValue, setSearchValue] = useState('');
  const [activeDir, setActiveDir] = useState('Все');
  const [activeLang, setActiveLang] = useState('Все');

  // Фильтруем проекты: только текущего автора + живой поиск + теги
  const myFilteredProjects = projects.filter(p => {
    const isOwn = p.author === user.username;
    const matchesSearch = p.title.toLowerCase().includes(searchValue.toLowerCase());
    const matchesDir = activeDir === 'Все' || p.direction === activeDir;
    const matchesLang = activeLang === 'Все' || p.lang.toLowerCase() === activeLang.toLowerCase();
    return isOwn && matchesSearch && matchesDir && matchesLang;
  });

  return (
    <div className="my-projects-wrapper">
      <div className="bg-glow glow-top-left"></div>
      <div className="bg-glow glow-bottom-right"></div>
      
      {/* 1. ШАПКА */}
      <Header 
        user={user} 
        searchValue={searchValue} 
        onSearchChange={setSearchValue} 
        setScreen={onBack} 
        projects={projects}
        onViewProjectDetails={onViewProjectDetails}
        ViewButtonMyProjects={false}
      />
      
      {/* 2. ОСНОВНАЯ СЕТКА GRID */}
      <div className="main-layout">
        
        {/* ЛЕВАЯ КОЛОНКА */}
        <aside className="back-row-margin">
          <button className="back-to-hub-btn" onClick={onBack}>
            ← Назад в ленту
          </button>
        </aside>

        {/* ЦЕНТРАЛЬНАЯ КОЛОНКА */}
        <div className="center-content-flow">
          <ProjectFilter 
            activeDir={activeDir} setActiveDir={setActiveDir} 
            activeLang={activeLang} setActiveLang={setActiveLang} 
          />

          <div className="projects-list-container">
            {myFilteredProjects.length > 0 ? (
              myFilteredProjects.map(project => (
                <div key={project.id} className="my-project-card-container" style={{ position: 'relative' }}>
                  {/* Рендерим безопасный компонент без хуков внутри цикла */}
                  <ProjectCard 
                    project={project} 
                    user={user} 
                    onViewDetails={onViewProjectDetails} 
                    onToggleLike={onToggleLike}
                  />
                  
                  {/* Кнопка "Удалить" аккуратно накладывается поверх нижней панели карточки */}
                  <button 
                    className="comment-trash-delete-btn" 
                    style={{ 
                      position: 'absolute', 
                      bottom: '28px', 
                      left: '180px', // Выравнивание кнопки удаления рядом с "Подробнее"

                      fontSize: '0.9rem',

                        onDeleteProject(project.id);
                      }

                    Удалить

                </div>

            ) : (

function RegisterPage({ onBack, onSubmitSuccess }) {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [group, setGroup] = useState('');
  const [about, setAbout] = useState('');
  const [isLoading, setIsLoading] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!username || !password || !group) {
      alert('Пожалуйста, заполните обязательные поля!');
      return;
    }

    setIsLoading(true);
    try {
      // 1. Шаг первый — шлем запрос на регистрацию
      const regResponse = await fetch('/api/auth/register', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username, password, group_name: group, about })
      });

      const regData = await regResponse.json();

      if (!regResponse.ok) {
        throw new Error(regData.detail || 'Ошибка при регистрации');
      }

      // 2. Шаг второй — если регистрация прошла успешно, сразу автоматически делаем логин
      const loginResponse = await fetch('/api/auth/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username, password })
      });

      const loginData = await loginResponse.json();

      if (loginResponse.ok) {
        localStorage.setItem('token', loginData.access_token);
        localStorage.setItem('user', JSON.stringify(loginData.user));
        onSubmitSuccess(loginData.user);
      } else {
        alert('Аккаунт создан, но автоматический вход не удался. Попробуйте войти вручную.');
        onBack();
      }

    } catch (error) {
      alert(error.message);
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="register-page-wrapper">
      <div className="bg-glow glow-top-left"></div>
      <div className="bg-glow glow-bottom-right"></div>

      <div className="register-page-container">
        <h1 className="auth-title">Chrt-hub</h1>

        <form className="register-form" onSubmit={handleSubmit}>
          <div className="form-group">
            <label>Имя пользователя *</label>
            <input 
              type="text" 
              placeholder="Придумайте логин" 
              value={username}
              onChange={(e) => setUsername(e.target.value)}
              disabled={isLoading}
            />
          </div>

          <div className="form-group">
            <label>Пароль *</label>
            <input 
              type="password" 
              placeholder="Минимум 6 символов" 
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              disabled={isLoading}
            />
          </div>

          <div className="form-group">
            <label>Учебная группа *</label>
            <select value={group} onChange={(e) => setGroup(e.target.value)} disabled={isLoading}>
              <option value="">Выберите вашу группу</option>
              {CHRT_GROUPS.map((g) => (
                <option key={g.value} value={g.value}>{g.label}</option>
              ))}
            </select>
          </div>

          <div className="form-group">
            <label>О себе (необязательно)</label>
            <textarea 
              placeholder="Расскажите о своих навыках или стеке технологий..." 
              value={about}
              onChange={(e) => setAbout(e.target.value)}
              disabled={isLoading}

import { useState, useEffect } from 'react';
import './IntroAnim.css';

function IntroAnim({ onAnimComplete }) {
  const [status, setStatus] = useState(''); // '', 'fade-text', 'slide-out'

  useEffect(() => {
    // Шаг 1: Ждем пока текст прилетит и посветит (2.4 секунды)
    const textTimer = setTimeout(() => {
      setStatus('fade-text');
    }, 1700);

    // Шаг 2: Через 400мс после затухания текста раздвигаем створки экрана
    const gateTimer = setTimeout(() => {
      setStatus('fade-text slide-out');
    }, 2100);

    const fadeScreenTimer = setTimeout(() => {
      setStatus('fade-text slide-out fade-out-screen');
    }, 2700);

    // Шаг 3: Полностью выгружаем интро из памяти, когда створки улетели за экран (1.2 секунды на анимацию)
    const completeTimer = setTimeout(() => {
      onAnimComplete();
    }, 4000);

    return () => {
      clearTimeout(textTimer);
      clearTimeout(gateTimer);
      clearTimeout(completeTimer);
    };
  }, [onAnimComplete]);

  return (
    <div className={`intro-overlay ${status}`}>
      {/* Левая створка */}
      <div className="intro-gate gate-left"></div>
      
      {/* Неоновый текст по центру */}
      <div className="intro-text-box">
        <h1 className="intro-neon-logo">chrt-hub</h1>
      </div>
      
      {/* Правая створка */}
      <div className="intro-gate gate-right"></div>
    </div>
  );
}

export default IntroAnim;

          user={user}
          onBack={() => setCurrentScreen('main_page')}
          onViewProjectDetails={(nextProject) => setSelectedProject(nextProject)} 
          onOpenMyProjects={() => setCurrentScreen('my_projects')}
        />
      )}

      {!showIntro && currentScreen === 'my_projects' && (
        <MyProjectsPage 
          user={user} 
          projects={projects} 
          onBack={() => setCurrentScreen('main_page')} 
          onDeleteProject={handleApiDeleteProject}
          onViewProjectDetails={(project) => {
            setSelectedProject(project);
            setCurrentScreen('project_detail');
          }}
        />
      )}
    </>
  );
}

export default App;
