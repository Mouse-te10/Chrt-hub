import { useState, useRef } from 'react';
import './ProjectForm.css';

function ProjectForm({ onAddProject, authorName }) {
  const [title, setTitle] = useState('');
  const [shortDesc, setShortDesc] = useState('');
  const [fullDesc, setFullDesc] = useState('');
  const [direction, setDirection] = useState('Веб-разработка');
  const [lang, setLang] = useState('Python');
  const [github, setGithub] = useState('');
  const [disableComments, setDisableComments] = useState(false);
  
  // Состояние для хранения выбранных медиафайлов
  const [mediaFiles, setMediaFiles] = useState([]);
  const fileInputRef = useRef(null);

  const handleFileChange = (e) => {
    const files = Array.from(e.target.files);
    
    // Проверка 1: Лимит на общее количество (до 10 штук суммарно)
    if (mediaFiles.length + files.length > 10) {
      alert('Максимальное количество медиафайлов (фото + видео) — 10 штук!');
      return;
    }

    files.forEach(file => {
      // Проверка 2: Если это видео, проверяем его длину
      if (file.type.startsWith('video/')) {
        const videoElement = document.createElement('video');
        videoElement.preload = 'metadata';
        
        videoElement.onloadedmetadata = () => {
          window.URL.revokeObjectURL(videoElement.src);
          // Длина видео не должна превышать 30 секунд
          if (videoElement.duration > 30) {
            alert(`Видео "${file.name}" превышает лимит в 30 секунд! Загрузка отменена.`);
          } else {
            addFileToState(file);
          }
        };
        videoElement.src = URL.createObjectURL(file);
      } else if (file.type.startsWith('image/')) {
        // Если это фото, добавляем сразу без проверки времени
        addFileToState(file);
      } else {
        alert('Разрешено загружать только фотографии и видеофайлы!');
      }
    });
  };

  const addFileToState = (file) => {
    const fileObj = {
      file,
      preview: URL.createObjectURL(file), // Создаем временную ссылку для отображения на экране
      type: file.type.startsWith('video/') ? 'video' : 'image'
    };
    setMediaFiles(prev => [...prev, fileObj]);
  };

  const removeFile = (index) => {
    setMediaFiles(prev => prev.filter((_, i) => i !== index));
  };

  const handleFormSubmit = (e) => {
    e.preventDefault();
    if (!title || !shortDesc) {
      alert('Заполните название и краткое описание!');
      return;
    }

    onAddProject({
      title: title,
      shortDesc: shortDesc,
      fullDesc: fullDesc || shortDesc,
      direction: direction,
      lang: direction === 'Дизайн' ? 'Дизайн' : lang,
      github: direction === 'Дизайн' ? '' : github || 'https://github.com',
      commentsDisabled: disableComments,
      mediaFiles: mediaFiles // Передаем наверх весь исходный массив со структурами { file, preview, type }
    });

    // Полная очистка формы
    setTitle('');
    setShortDesc('');
    setFullDesc('');
    setGithub('');
    setMediaFiles([]);
  };



  return (
    <aside className="project-form-sidebar">
      <h3 className="form-title">форма</h3>
      <form className="mini-form" onSubmit={handleFormSubmit}>
        <input type="text" placeholder="Название проекта" value={title} onChange={e => setTitle(e.target.value)} />
        <input type="text" placeholder="Краткое описание" value={shortDesc} onChange={e => setShortDesc(e.target.value)} />
        <textarea placeholder="Полное описание" rows="2" value={fullDesc} onChange={e => setFullDesc(e.target.value)}></textarea>
        <input type="text" placeholder="Ссылка на GitHub (если есть)" value={github} onChange={e => setGithub(e.target.value)} disabled={direction === 'Дизайн'} />
        
        <select value={direction} onChange={e => setDirection(e.target.value)}>
          <option value="Веб-разработка">Веб-разработка</option>
          <option value="Мобильная разработка">Мобильная разработка</option>
          <option value="Разработка программ">Разработка программ</option>
          <option value="Game-dev">Game-dev</option>
          <option value="Скрипты">Скрипты</option>
          <option value="Дизайн">Дизайн</option>
          <option value="Другое">Другое</option>
        </select>

        <select value={lang} onChange={e => setLang(e.target.value)} disabled={direction === 'Дизайн'}>
          <option value="Python">Python</option>
          <option value="Js">Js</option>
          <option value="Ts">Ts</option>
          <option value="Bash">Bash</option>
          <option value="C++">C++</option>
          <option value="C#">C#</option>
        </select>

        {/* НОВАЯ СЕКЦИЯ: Загрузка медиафайлов */}
        <div className="media-upload-section">
          <button type="button" className="upload-trigger-btn" onClick={() => fileInputRef.current.click()}>
            📎 Прикрепить фото/видео ({mediaFiles.length}/10)
          </button>
          <input 
            type="file" 
            ref={fileInputRef} 
            style={{ display: 'none' }} 
            multiple 
            accept="image/*,video/*" 
            onChange={handleFileChange} 
          />
          
          {/* Сетка превьюшек */}
          {mediaFiles.length > 0 && (
            <div className="form-media-preview-grid">
              {mediaFiles.map((file, idx) => (
                <div key={idx} className="preview-item-box">
                  {file.type === 'video' ? (
                    <video src={file.preview} muted />
                  ) : (
                    <img src={file.preview} alt="" />
                  )}
                  <button type="button" className="delete-media-btn" onClick={() => removeFile(idx)}>×</button>
                </div>
              ))}
            </div>
          )}
        </div>

        <label className="checkbox-group">
          <input type="checkbox" checked={disableComments} onChange={e => setDisableComments(e.target.checked)} />
          комментарии отключить
        </label>

        <button type="submit" className="submit-project-btn">отправить</button>
      </form>
    </aside>
  );
}

export default ProjectForm;

import React, { useState, useEffect } from 'react';
import { projectService } from '../../../services/projectService';
import './SidebarRating.css';

function SidebarRating({ refreshTrigger }) {
  const [leaders, setLeaders] = useState([]);
  const [loading, setLoading] = useState(true);

  // Подгружаем рейтинг при монтировании и при триггере (например, создании/удалении проекта)
  useEffect(() => {
    const loadRating = async () => {
      try {
        const data = await projectService.getLeaderboard();
        setLeaders(data);
      } catch (err) {
        console.error('Ошибка рейтинга:', err.message);
      } finally {
        setLoading(false);
      }
    };
    loadRating();
  }, [refreshTrigger]);

  return (
    <aside className="rating-sidebar">
      <h3 className="rating-title">Рейтинг</h3>
      <div className="rating-list">
        {loading ? (
          <div style={{ color: 'var(--text-muted)', fontSize: '0.9rem' }}>Загрузка топа...</div>
        ) : leaders.length > 0 ? (
          leaders.map((student, i) => (
            <div key={i} className="rating-item">
              <span className="user-name">{student.name}</span>
              <span className="user-score">{student.count}</span>
            </div>
          ))
        ) : (
          <div style={{ color: 'var(--text-muted)', fontSize: '0.9rem' }}>Пусто</div>
        )}
      </div>
    </aside>
  );
}

export default SidebarRating;

import { useState, useRef, useEffect } from 'react';
import './Header.css';

// Добавили пропс showDropdown в аргументы функции
function Header({ user, onSearchChange, searchValue, setScreen, projects, onViewProjectDetails, showDropdown = false, ViewButtonMyProjects }) {
  const [isDropdownOpen, setIsDropdownOpen] = useState(false);
  const dropdownRef = useRef(null);

  // Фильтруем проекты для подсказок только если включен флаг showDropdown
  const searchSuggestions = showDropdown && projects && searchValue.trim() !== ''
    ? projects.filter(p => p.title.toLowerCase().includes(searchValue.toLowerCase()))
    : [];

  useEffect(() => {
    function handleClickOutside(event) {
      if (dropdownRef.current && !dropdownRef.current.contains(event.target)) {
        setIsDropdownOpen(false);
      }
    }
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  const handleSuggestionClick = (project) => {
    setIsDropdownOpen(false);
    onSearchChange(''); 
    if (onViewProjectDetails) {
      onViewProjectDetails(project); 
    }
  };

  return (
    <header className="main-header">
      <div className="header-left">
        {ViewButtonMyProjects && (
          user.role !== 'anonymous' ? (
            <button className="my-projects-btn" onClick={() => setScreen('my_projects')}>
              Мои проекты
            </button>
          ) : (
            <span className='user-badge'>Аноним</span>
          )
        )}

        
        
        <div className="search-container" ref={dropdownRef}>
          <input 
            type="text" 
            className="search-input" 
            placeholder="Быстрый поиск проектов..." 
            value={searchValue}
            onChange={(e) => {
              onSearchChange(e.target.value);
              setIsDropdownOpen(true);
            }}
            onFocus={() => setIsDropdownOpen(true)}
          />

          {/* МЕНЮ ВЫЕДЕТ ТОЛЬКО ЕСЛИ ОНО РАЗРЕШЕНО ФЛАГОМ */}
          {showDropdown && isDropdownOpen && searchSuggestions.length > 0 && (
            <div className="search-autocomplete-dropdown">
              {searchSuggestions.map(project => (
                <div 
                  key={project.id} 
                  className="suggestion-item"
                  onClick={() => handleSuggestionClick(project)}
                >
                  <div className="suggestion-info">
                    <span className="suggestion-title">{project.title}</span>
                    <span className="suggestion-author">от {project.author}</span>
                  </div>
                  <span className="suggestion-tag">{project.direction}</span>
                </div>
              ))}
            </div>
          )}

          {/* ЗАГЛУШКА ВЫЕДЕТ ТОЛЬКО ЕСЛИ ОНО РАЗРЕШЕНО ФЛАГОМ */}
          {showDropdown && isDropdownOpen && searchValue.trim() !== '' && searchSuggestions.length === 0 && (
            <div className="search-autocomplete-dropdown empty-result">
              Ничего не найдено
            </div>
          )}
        </div>
      </div>
{/* ... далее весь код кнопок авторизации в Header.jsx идет без изменений ... */}


      <div className="header-right">
        {user.role === 'anonymous' ? (
          <>
            
            <button className="auth-nav-btn" onClick={() => setScreen('login')}>Войти</button>
            <button className="auth-nav-btn" onClick={() => setScreen('register')}>Регистрация</button>
          </>
        ) : (
          <>
            <span className="user-badge">{user.username} <span>{user.projectCount}</span></span>
            <button className="logout-btn" onClick={() => setScreen('auth_choice')}>Выйти</button>
          </>
        )}
      </div>
    </header>
  );

import { useState } from 'react';
import './LoginPage.css';

function LoginPage({ onBack, onSubmitSuccess }) {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [isLoading, setIsLoading] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!username || !password) {
      alert('Заполните все поля!');
      return;
    }

    setIsLoading(true);
    try {
      const response = await fetch('http://localhost:8000/api/auth/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username, password })
      });

      const data = await response.json();

      if (!response.ok) {
        // Если бэкенд вернул ошибку (401 и т.д.), выводим её текст
        throw new Error(data.detail || 'Ошибка при входе');
      }

      // Сохраняем JWT-токен в localStorage, чтобы сессия не сбрасывалась
      localStorage.setItem('token', data.access_token);
      localStorage.setItem('user', JSON.stringify(data.user));

      // Передаем данные наверх в App.jsx
      onSubmitSuccess(data.user);
    } catch (error) {
      alert(error.message);
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="login-page-wrapper">
      <div className="bg-glow glow-top-left"></div>
      <div className="bg-glow glow-bottom-right"></div>

      <div className="login-page-container">
        <h1 className="auth-title">Chrt-hub</h1>

        <form className="register-form" onSubmit={handleSubmit}>
          <div className="form-group">
            <label>Имя пользователя</label>
            <input 
              type="text" 
              placeholder="Введите логин" 
              value={username}
              onChange={(e) => setUsername(e.target.value)}
              disabled={isLoading}
            />
          </div>

          <div className="form-group">
            <label>Пароль</label>
            <input 
              type="password" 
              placeholder="Введите ваш пароль" 
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              disabled={isLoading}
            />
          </div>

          <button type="submit" className="submit-btn" disabled={isLoading}>
            {isLoading ? 'Вход...' : 'Войти'}
          </button>
          
          <button type="button" className="back-btn" onClick={onBack} disabled={isLoading}>
            ← Назад к выбору
          </button>
        </form>
      </div>
    </div>
  );
}

export default LoginPage;

import { useState, useEffect } from 'react';
import './ProjectPage.css';
import Header from '../MainPage/Header/Header';
import Footer from '../MainPage/Footer/Footer';
import { projectService } from '../../services/projectService'; // Подключаем сервис

function ProjectPage({ project, currentUsername, onBack, projects, onViewProjectDetails, onOpenMyProjects, user }) {
  const [mediaIndex, setMediaIndex] = useState(0);
  const [isLightboxOpen, setIsLightboxOpen] = useState(false);
  const [pageSearchValue, setPageSearchValue] = useState('');

  // Заменяем хардкод на динамический стейт из БД
  const [comments, setComments] = useState([]);
  const [newCommentText, setNewCommentText] = useState('');
  const [commentsLoading, setCommentsLoading] = useState(true);

  const projectMedia = project?.media && project.media.length > 0 
    ? project.media 
    : ["https://unsplash.com"];

  // Подгружаем комментарии при монтировании страницы или смене ID проекта
  useEffect(() => {
    if (project?.id) {
      loadComments();
    }
  }, [project?.id]);

  const loadComments = async () => {
    try {
      setCommentsLoading(true);
      const data = await projectService.getComments(project.id);
      setComments(data);
    } catch (err) {
      console.error(err.message);
    } finally {
      setCommentsLoading(false);
    }
  };

  const nextMedia = () => setMediaIndex((prev) => (prev === projectMedia.length - 1 ? 0 : prev + 1));
  const prevMedia = () => setMediaIndex((prev) => (prev === 0 ? projectMedia.length - 1 : prev - 1));

  // Реальное голосование через API с оптимистичным обновлением интерфейса
  const handleVote = async (commentId, voteType) => {
    if (user.role === 'anonymous') {
      alert('Голосовать могут только авторизованные студенты ЧРТ!');
      return;
    }

    const isLikeTarget = voteType === 'like';
    
    try {
      // Отправляем запрос на бэк
      await projectService.voteComment(commentId, isLikeTarget);
      // Просто пересобираем комменты из БД, чтобы получить идеальные точные счетчики
      const updatedComments = await projectService.getComments(project.id);
      setComments(updatedComments);
    } catch (err) {
      alert(err.message);
    }
  };

  // Добавление комментария через API
  const handleAddComment = async (e) => {
    e.preventDefault();
    if (user.role === 'anonymous') {
      alert('Анонимы не могут оставлять комментарии. Пожалуйста, войдите в аккаунт!');
      return;
    }
    if (!newCommentText.trim()) return;

    try {
      const createdComment = await projectService.createComment(newCommentText, project.id);
      // Добавляем новый коммент в список локально
      setComments([...comments, createdComment]);
      setNewCommentText('');
    } catch (err) {
      alert(err.message);
    }
  };

  const isDesign = project?.direction === 'Дизайн';

  return (
    <div className="project-page-root" style={{backgroundColor: 'var(--bg-main)', minHeight: '100vh'}}>
      <div className="bg-glow glow-top-left"></div>
      <div className="bg-glow glow-bottom-right"></div>
      <Header 
        user={user} 
        searchValue={pageSearchValue} 
        onSearchChange={setPageSearchValue} 
        setScreen={(targetScreen) => {
          if (targetScreen === 'my_projects') {
            onOpenMyProjects();
          } else {
            onBack();
          }
        }}
        projects={projects} 
        onViewProjectDetails={onViewProjectDetails} 
        showDropdown={true}
        ViewButtonMyProjects={true}
      />

      <div className="project-sub-header">
        <button className="back-to-hub-btn" onClick={onBack}>
          ← Назад в ленту
        </button>
        {isDesign ? (
          <button className="github-top-btn disabled" disabled>Код недоступен (Дизайн)</button>
        ) : (
          <a href={project?.github || "https://github.com"} target="_blank" rel="noreferrer" className="github-top-btn">
            Перейти на GitHub
          </a>
        )}
      </div>

      <div className="project-details-grid">
        <main className="project-content-left">
          <h2 className="project-main-title">{project?.title || "Название проекта"}</h2>
          <p className="project-long-desc">{project?.fullDesc || project?.shortDesc || "Описание проекта..."}</p>

          <div className="gallery-slider-container">
            <button className="slider-arrow left" onClick={prevMedia}>‹</button>
            <div className="slider-media-window" onClick={() => setIsLightboxOpen(true)}>
              {projectMedia[mediaIndex].includes('video') || (typeof projectMedia[mediaIndex] === 'string' && projectMedia[mediaIndex].endsWith('.mp4')) ? (
                <video src={projectMedia[mediaIndex]} muted autoPlay loop />
              ) : (
                <img src={projectMedia[mediaIndex]} alt="Скриншот проекта" />
              )}
            </div>
            <button className="slider-arrow right" onClick={nextMedia}>›</button>
          </div>

          {/* Блок Комментариев */}
          <section className="project-comments-wrapper">
            <h3 className="comments-block-title">Комментарии ({comments.length})</h3>
            
            {project?.commentsDisabled ? (
              <div style={{color: 'var(--text-muted)', fontStyle: 'italic', padding: '10px 0'}}>
                Автор отключил комментарии к этому проекту.
              </div>
            ) : commentsLoading ? (
              <div style={{color: 'var(--text-muted)'}}>Загрузка обсуждения...</div>
            ) : (
              <>
                <div className="comments-flow-list">
                  {comments.map((c) => (
                    <div key={c.id} className="comment-node-box">
                      <div className="comment-meta-row">
                        <span>
                          {c.author_name} ({c.author_group}) {c.author_name === currentUsername && <span style={{color: 'var(--accent-mint)'}}>(Вы)</span>}
                        </span>
                      </div>
                      <p className="comment-text-body">{c.text}</p>
                      
                      <div className="comment-actions-bar">
                        <div className="voting-buttons-pack">
                          <button 
                            type="button"
                            className={`vote-action-btn liked ${c.user_vote === true ? 'active' : ''}`}
                            onClick={() => handleVote(c.id, 'like')}
                          >
                            ▲ {c.likes_count}
                          </button>
                          <button 
                            type="button"
                            className={`vote-action-btn disliked ${c.user_vote === false ? 'active' : ''}`}
                            onClick={() => handleVote(c.id, 'dislike')}
                          >
                            ▼ {c.dislikes_count}
                          </button>
                        </div>
                      </div>
                    </div>
                  ))}
                  {comments.length === 0 && (
                    <div style={{color: 'var(--text-muted)', padding: '10px 0'}}>Будьте первым, кто оставит отзыв!</div>
                  )}
                </div>

                {user.role !== 'anonymous' && (
                  <form className="add-comment-form-block" onSubmit={handleAddComment}>
                    <input 
                      type="text" 
                      placeholder="Напишите ваш отзыв или вопрос по коду..." 
                      value={newCommentText}
                      onChange={e => setNewCommentText(e.target.value)}
                      maxLength={1000}
                    />
                    <button type="submit">Отправить</button>
                  </form>
                )}
              </>
            )}
          </section>
        </main>

        <aside className="project-author-sidebar">
          <div className="sidebar-heading">Автор проекта</div>
          <div className="author-main-profile">
            <div className="author-huge-name">{project?.author || "Студент ЧРТ"}</div>
            <div className="author-group-tag">Группа: {project?.group_name || "ЧРТ"}</div>
          </div>
          <div className="author-rating-score-box">
            <span style={{ color: 'var(--text-muted)' }}>Рейтинг на хабе:</span>
            <span style={{ color: 'var(--accent-mint)', fontWeight: 700 }}>★ 100</span>
          </div>
          <div className="author-bio-text">

          </div>
        </aside>
      </div>

      {isLightboxOpen && (
        <div className="lightbox-modal-overlay" onClick={() => setIsLightboxOpen(false)}>
          <div className="lightbox-content-box" onClick={e => e.stopPropagation()}>
            <button type="button" className="lightbox-close-top-btn" onClick={() => setIsLightboxOpen(false)}>
              ← Назад
            </button>
            {projectMedia[mediaIndex].includes('video') || (typeof projectMedia[mediaIndex] === 'string' && projectMedia[mediaIndex].endsWith('.mp4')) ? (
              <video src={projectMedia[mediaIndex]} controls autoPlay />

              <img src={projectMedia[mediaIndex]} alt="Увеличенное превью" />

import { useState } from 'react';
import Header from '../MainPage/Header/Header';
import Footer from '../MainPage/Footer/Footer';
import ProjectFilter from '../MainPage/ProjectFilter/ProjectFilter';
import './MyProjectsPage.css';

function MyProjectsPage({ user, projects, onBack, onDeleteProject, onViewProjectDetails }) {
  const [searchValue, setSearchValue] = useState('');
  const [activeDir, setActiveDir] = useState('Все');
  const [activeLang, setActiveLang] = useState('Все');

  // Фильтруем проекты: только текущего автора + живой поиск + теги
  const myFilteredProjects = projects.filter(p => {
    const isOwn = p.author === user.username;
    const matchesSearch = p.title.toLowerCase().includes(searchValue.toLowerCase());
    const matchesDir = activeDir === 'Все' || p.direction === activeDir;
    const matchesLang = activeLang === 'Все' || p.lang.toLowerCase() === activeLang.toLowerCase();
    return isOwn && matchesSearch && matchesDir && matchesLang;
  });

  return (
    <div className="my-projects-wrapper">
      <div className="bg-glow glow-top-left"></div>
      <div className="bg-glow glow-bottom-right"></div>
      {/* 1. СТАНДАРТНАЯ ШАПКА ИЗ MAIN PAGE */}
      <Header 
        user={user} 
        searchValue={searchValue} 
        onSearchChange={setSearchValue} 
        setScreen={onBack} 
        projects={projects}
        onViewProjectDetails={onViewProjectDetails}
        ViewButtonMyProjects={false}
      />
      
      {/* 2. СТАНДАРТНАЯ СЕТКА GRID ИЗ MAIN PAGE (.main-layout) */}
      <div className="main-layout">
        
        {/* ЛЕВАЯ КОЛОНКА: Кнопка назад вместо рейтинга */}
        <aside className="back-row-margin">
          <button className="back-to-hub-btn" onClick={onBack}>
            ← Назад в ленту
          </button>
        </aside>

        {/* ЦЕНТРАЛЬНАЯ КОЛОНКА: Лента фильтров и блоки проектов из MainPage */}
        <div className="center-content-flow">
          <ProjectFilter 
            activeDir={activeDir} setActiveDir={setActiveDir} 
            activeLang={activeLang} setActiveLang={setActiveLang} 
          />

          <div className="projects-list-container">
            {myFilteredProjects.length > 0 ? (
              myFilteredProjects.map(project => {
                const isDesign = project.direction === 'Дизайн';
                return (
                  // Используем структуру оригинальной карточки проекта
                  <article key={project.id} className="project-card-main">
                    <div className="card-header-row">
                      <h4 className="card-title">{project.title}</h4>
                      <div className="card-tags">
                        {!isDesign && <span className="badge lang">{project.lang}</span>}
                        <span className="badge dir">{project.direction}</span>
                      </div>
                    </div>

                    <p className="card-desc">{project.shortDesc}</p>

                    <div className="card-footer-row">
                      <div className="card-author">Автор: <span>{project.author}</span></div>
                      <div className="card-actions">
                        <button className="details-btn" onClick={() => onViewProjectDetails(project)}>
                          Подробнее
                        </button>
                        <button 
                          className="comment-trash-delete-btn" 
                          style={{ marginLeft: '10px' }}
                          onClick={() => {
                            if (window.confirm(`Удалить проект "${project.title}"?`)) {
                              onDeleteProject(project.id);
                            }
                          }}
                        >
                          Удалить
                        </button>
                      </div>
                    </div>
                  </article>
                );
              })
            ) : (
              <div style={{color: 'var(--text-muted)', textAlign: 'center', padding: '40px'}}>
                У вас пока нет проектов в этой категории.
              </div>
            )}
          </div>
        </div>

        {/* ПРАВАЯ КОЛОНКА: Сайдбар профиля из ProjectPage (.project-author-sidebar) */}
        <aside className="project-author-sidebar">
          <div className="sidebar-heading">Ваш профиль</div>
          
          <div className="author-main-profile">
            <div className="author-huge-name">{user.username}</div>
            <div className="author-group-tag">Студент ЧРТ</div>
          </div>

          <div className="author-rating-score-box">
            <span style={{ color: 'var(--text-muted)' }}>Ваш рейтинг:</span>
            <span style={{ color: 'var(--accent-mint)', fontWeight: 700 }}>★ {user.projectCount * 100}</span>
          </div>

          <div className="author-bio-text">
            {user.about || "Информация о навыках и стеке технологий не заполнена."}
          </div>
        </aside>

      </div>

      {/* 3. СТАНДАРТНЫЙ ПОДВАЛ ИЗ MAIN PAGE */}
      <Footer />
    </div>
  );
}

export default MyProjectsPage;

function RegisterPage({ onBack, onSubmitSuccess }) {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [group, setGroup] = useState('');
  const [about, setAbout] = useState('');
  const [isLoading, setIsLoading] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!username || !password || !group) {
      alert('Пожалуйста, заполните обязательные поля!');
      return;
    }

    setIsLoading(true);
    try {
      // 1. Шаг первый — шлем запрос на регистрацию
      const regResponse = await fetch('http://localhost:8000/api/auth/register', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username, password, group_name: group, about })
      });

      const regData = await regResponse.json();

      if (!regResponse.ok) {
        throw new Error(regData.detail || 'Ошибка при регистрации');
      }

      // 2. Шаг второй — если регистрация прошла успешно, сразу автоматически делаем логин
      const loginResponse = await fetch('http://localhost:8000/api/auth/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username, password })
      });

      const loginData = await loginResponse.json();

      if (loginResponse.ok) {
        localStorage.setItem('token', loginData.access_token);
        localStorage.setItem('user', JSON.stringify(loginData.user));
        onSubmitSuccess(loginData.user);
      } else {
        alert('Аккаунт создан, но автоматический вход не удался. Попробуйте войти вручную.');
        onBack();
      }

    } catch (error) {
      alert(error.message);
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="register-page-wrapper">
      <div className="bg-glow glow-top-left"></div>
      <div className="bg-glow glow-bottom-right"></div>

      <div className="register-page-container">
        <h1 className="auth-title">Chrt-hub</h1>

        <form className="register-form" onSubmit={handleSubmit}>
          <div className="form-group">
            <label>Имя пользователя *</label>
            <input 
              type="text" 
              placeholder="Придумайте логин" 
              value={username}
              onChange={(e) => setUsername(e.target.value)}
              disabled={isLoading}
            />
          </div>

          <div className="form-group">
            <label>Пароль *</label>
            <input 
              type="password" 
              placeholder="Минимум 6 символов" 
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              disabled={isLoading}
            />
          </div>

          <div className="form-group">
            <label>Учебная группа *</label>
            <select value={group} onChange={(e) => setGroup(e.target.value)} disabled={isLoading}>
              <option value="">Выберите вашу группу</option>
              {CHRT_GROUPS.map((g) => (
                <option key={g.value} value={g.value}>{g.label}</option>
              ))}
            </select>
          </div>

          <div className="form-group">
            <label>О себе (необязательно)</label>
            <textarea 
              placeholder="Расскажите о своих навыках или стеке технологий..." 
              value={about}
              onChange={(e) => setAbout(e.target.value)}
              disabled={isLoading}
            />
          </div>

          <button type="submit" className="submit-btn" disabled={isLoading}>
            {isLoading ? 'Создание аккаунта...' : 'Создать аккаунт'}
          </button>
          
          <button type="button" className="back-btn" onClick={onBack} disabled={isLoading}>
            ← Назад к выбору
          </button>
        </form>
      </div>
    </div>
  );
}

import { useState, useEffect } from 'react';
import './IntroAnim.css';

function IntroAnim({ onAnimComplete }) {
  const [status, setStatus] = useState(''); // '', 'fade-text', 'slide-out'

  useEffect(() => {
    // Шаг 1: Ждем пока текст прилетит и посветит (2.4 секунды)
    const textTimer = setTimeout(() => {
      setStatus('fade-text');
    }, 1700);

    // Шаг 2: Через 400мс после затухания текста раздвигаем створки экрана
    const gateTimer = setTimeout(() => {
      setStatus('fade-text slide-out');
    }, 2100);

    const fadeScreenTimer = setTimeout(() => {
      setStatus('fade-text slide-out fade-out-screen');
    }, 2700);

    // Шаг 3: Полностью выгружаем интро из памяти, когда створки улетели за экран (1.2 секунды на анимацию)
    const completeTimer = setTimeout(() => {
      onAnimComplete();
    }, 4000);

    return () => {
      clearTimeout(textTimer);
      clearTimeout(gateTimer);
      clearTimeout(completeTimer);
    };
  }, [onAnimComplete]);

  return (
    <div className={`intro-overlay ${status}`}>
      {/* Левая створка */}
      <div className="intro-gate gate-left"></div>
      
      {/* Неоновый текст по центру */}
      <div className="intro-text-box">
        <h1 className="intro-neon-logo">chrt-hub</h1>
      </div>
      
      {/* Правая створка */}
      <div className="intro-gate gate-right"></div>
    </div>
  );
}

export default IntroAnim;

const BASE_URL = 'http://localhost:8000/api';

const getAuthHeaders = () => {
  const token = localStorage.getItem('token'); // Синхронизировано с LoginPage/RegisterPage
  return token ? { 'Authorization': `Bearer ${token}` } : {};
};

export const projectService = {
  // Получить все проекты с учетом фильтров
  getAllProjects: async (direction = 'Все', lang = 'Все') => {
    const res = await fetch(`${BASE_URL}/projects?direction=${encodeURIComponent(direction)}&lang=${encodeURIComponent(lang)}`);
    if (!res.ok) throw new Error('Не удалось загрузить ленту Chrt-Hub');
    return await res.json();
  },

  // Получить проекты залогиненного студента
  getMyProjects: async () => {
    const res = await fetch(`${BASE_URL}/my/projects`, {
      headers: getAuthHeaders(),
    });
    if (!res.ok) throw new Error('Не удалось загрузить ваши проекты');
    return await res.json();
  },

    // Опубликовать проект (теперь с реальными файлами)
  createProject: async (projectData) => {
    const formData = new FormData();
    
    // Переносим текстовые поля формы в FormData
    formData.append('title', projectData.title);
    formData.append('shortDesc', projectData.shortDesc);
    formData.append('fullDesc', projectData.fullDesc);
    formData.append('direction', projectData.direction);
    formData.append('lang', projectData.lang);
    formData.append('github', projectData.github);
    formData.append('commentsDisabled', projectData.commentsDisabled);

    // Добавляем файлы из нашего локального стейта (если они прикреплены)
    if (projectData.mediaFiles && projectData.mediaFiles.length > 0) {
      projectData.mediaFiles.forEach(fileObj => {
        formData.append('files', fileObj.file); // Ключ 'files' должен совпадать с аргументом в FastAPI
      });
    }

    const token = localStorage.getItem('token');
    const headers = token ? { 'Authorization': `Bearer ${token}` } : {};

    const res = await fetch(`${BASE_URL}/projects`, {
      method: 'POST',
      headers: headers, // Браузер сам выставит нужный Boundary для multipart/form-data, Content-Type руками писать НЕ НАДО
      body: formData,
    });

    if (!res.ok) {
      const err = await res.json();
      throw new Error(err.detail || 'Ошибка при публикации проекта с медиа');
    }
    return await res.json();
  },

  // Удалить проект
  deleteProject: async (id) => {
    const res = await fetch(`${BASE_URL}/projects/${id}`, {
      method: 'DELETE',
      headers: getAuthHeaders(),
    });
    if (!res.ok) {
      const err = await res.json();
      throw new Error(err.detail || 'Не удалось удалить проект');
    }
    return true;
  },
    // Получить комментарии к проекту
  getComments: async (projectId) => {
    const token = localStorage.getItem('token');
    const headers = token ? { 'Authorization': `Bearer ${token}` } : {};
    const res = await fetch(`${BASE_URL}/projects/${projectId}/comments`, { headers });
    if (!res.ok) throw new Error('Не удалось загрузить комментарии');
    return await res.json();
  },

  // Создать комментарий
  createComment: async (text, projectId) => {
    const res = await fetch(`${BASE_URL}/comments`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        ...getAuthHeaders(),
      },
      body: JSON.stringify({ text, project_id: projectId }),
    });
    if (!res.ok) {
      const err = await res.json();
      throw new Error(err.detail || 'Не удалось отправить комментарий');
    }
    return await res.json();
  },

  // Проголосовать (лайк/дизлайк)
  voteComment: async (commentId, isLike) => {
    const res = await fetch(`${BASE_URL}/comments/${commentId}/vote`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        ...getAuthHeaders(),
      },
      body: JSON.stringify({ is_like: isLike }),
    });
    if (!res.ok) {
      const err = await res.json();
      throw new Error(err.detail || 'Не удалось проголосовать');
    }
    return await res.json();
  },
    // Получить ТОП-5 студентов по количеству проектов
  getLeaderboard: async () => {
    const res = await fetch(`${BASE_URL}/users/rating`);
    if (!res.ok) throw new Error('Не удалось загрузить рейтинг ЧРТ');
    return await res.json();
  }
};

function App() {
  // Экраны: 'welcome' | 'auth_choice' | 'login' | 'register' | 'main_page'
  const [currentScreen, setCurrentScreen] = useState('welcome');
  const [currentIndex, setCurrentIndex] = useState(0);

  const [user, setUser] = useState(() => {
    const savedUser = localStorage.getItem('user');
    return savedUser ? JSON.parse(savedUser) : { role: 'anonymous', username: 'аноним', projectCount: 0 };
  });

  const [projects, setProjects] = useState([]);
  const [selectedProject, setSelectedProject] = useState(null); 
  const [showIntro, setShowIntro] = useState(true);

  const [activeDir, setActiveDir] = useState('Все');
  const [activeLang, setActiveLang] = useState('Все');

  
  useEffect(() => {
    const token = localStorage.getItem('token');
    const savedUser = localStorage.getItem('user');
    if (token && savedUser) {
      setUser(JSON.parse(savedUser));
      setCurrentScreen('main_page');
    }
  }, []);

  useEffect(() => {
    if (currentScreen === 'main_page' || currentScreen === 'project_detail' || currentScreen === 'my_projects') {
      const loadAllProjects = async () => {
        try {
          const data = await projectService.getAllProjects(activeDir, activeLang);
          setProjects(data);
          
          // Актуализируем счетчик собственных проектов у вошедшего юзера
          if (user.role !== 'anonymous') {
            const myCount = data.filter(p => p.author === user.username).length;
            setUser(prev => ({ ...prev, projectCount: myCount }));
          }
        } catch (err) {
          console.error(err.message);
        }
      };
      loadAllProjects();
    }
  }, [currentScreen, activeDir, activeLang, user.username]);

  const handleAuthSuccess = (userData) => {
    const loggedUser = {
      role: 'student',
      username: userData.username,
      projectCount: 0, 
      about: userData.about || '',
      group_name: userData.group_name
    };
    setUser(loggedUser);
    localStorage.setItem('user', JSON.stringify(loggedUser)); // Перезаписываем со структурой ролей
    setCurrentScreen('main_page');
  };

  const handleLogout = () => {
    localStorage.removeItem('token');
    localStorage.removeItem('user');
    setUser({ role: 'anonymous', username: 'аноним', projectCount: 0 });
    setCurrentScreen('auth_choice');
  };
  
  
  // Слайдер (работает только на приветственном экране)
  useEffect(() => {
    if (currentScreen !== 'welcome') return;
    const interval = setInterval(() => {
      setCurrentIndex((prevIndex) => {
        const maxIndex = CARDS_DATA.length - 3;
        return prevIndex >= maxIndex ? 0 : prevIndex + 1;
      });
    }, 4000);
    return () => clearInterval(interval);
  }, [currentScreen]);

  // Функция для увеличения счетчика проектов у вошедшего студента
  const updateUserProjectCount = () => {
    setUser(prev => ({ ...prev, projectCount: prev.projectCount + 1 }));
  };

  // Переход 2: Студент нажал кнопку "Аноним"
  const handleAnonymousTransition = () => {
    setUser({ role: 'anonymous', username: 'аноним', projectCount: 0 });
    setCurrentScreen('main_page');
  };

  const handleAddProject = async (newProjectData) => {
    try {
      const savedProject = await projectService.createProject(newProjectData);
      setProjects(prev => [savedProject, ...prev]);
      setUser(prev => ({ ...prev, projectCount: prev.projectCount + 1 }));
      alert('Проект успешно опубликован на Chrt-Hub!');
    } catch (err) {
      alert(err.message);
    }
  };

  // Метод удаления проекта через API
  const handleApiDeleteProject = async (id) => {
    try {
      await projectService.deleteProject(id);
      setProjects(prev => prev.filter(p => p.id !== id));
      setUser(prev => ({ ...prev, projectCount: Math.max(0, prev.projectCount - 1) }));
    } catch (err) {
      alert(err.message);
    }
  };

  // Таймер приветственного слайдера
  useEffect(() => {
    if (currentScreen !== 'welcome') return;
    const interval = setInterval(() => {
      setCurrentIndex((prevIndex) => {
        const maxIndex = CARDS_DATA.length - 3;
        return prevIndex >= maxIndex ? 0 : prevIndex + 1;
      });
    }, 4000);
    return () => clearInterval(interval);
  }, [currentScreen]);

  return (
    <>
      {showIntro && <IntroAnim onAnimComplete={() => setShowIntro(false)} />}
      {/* 1. ЭКРАН ПРИВЕТСТВИЯ */}
      {!showIntro && currentScreen === 'welcome' && (
        <main className="welcome-container">
          <h1 className="welcome-title">Chrt-hub</h1>
          <p className="welcome-description">
            Локальная форма для студентов и преподавателей Челябинского Радиотехнического Техникума. 
            Публикуйте свои лучшие веб-сервисы, десктопные приложения, игры и дизайн-концепты.
          </p>
          <button className="start-button" onClick={() => setCurrentScreen('auth_choice')}>
            Начать!
          </button>
          
          <div className="slider-wrapper">
            <div 
              className="slider-track"
              style={{ transform: `translateX(calc(-${currentIndex * (100 / 3)}% - ${currentIndex * 8}px))` }}
            >
              {CARDS_DATA.map((card) => (
                <article key={card.id} className="preview-card">
                  <h3>{card.title}</h3>
                  <p>{card.desc}</p>
                </article>
              ))}
            </div>
          </div>
        </main>
      )}

      {/* 2. ЭКРАН ВЫБОРА (Вход / Регистрация / Аноним) */}
      {!showIntro && currentScreen === 'auth_choice' && (
        <AuthPage 
          onLoginClick={() => setCurrentScreen('login')}
          onRegisterClick={() => setCurrentScreen('register')}
          onAnonymousClick={handleAnonymousTransition}
        />
      )}

      {!showIntro && currentScreen === 'login' && (
        <LoginPage onBack={() => setCurrentScreen('auth_choice')} onSubmitSuccess={handleAuthSuccess} />
      )}

      {!showIntro && currentScreen === 'register' && (
        <RegisterPage onBack={() => setCurrentScreen('auth_choice')} onSubmitSuccess={handleAuthSuccess} />
      )}

      {!showIntro && currentScreen === 'main_page' && (
        <MainPage 
          user={user} 
          setScreen={(screen) => screen === 'auth_choice' ? handleLogout() : setCurrentScreen(screen)}
          projects={projects}
          // Прокидываем управление фильтрами наверх в App.jsx для триггера useEffect fetch-запросов
          activeDir={activeDir} setActiveDir={setActiveDir}
          activeLang={activeLang} setActiveLang={setActiveLang}
          onAddProject={handleAddProject}
          onViewProjectDetails={(project) => {
            setSelectedProject(project);
            setCurrentScreen('project_detail');
          }}
        />
      )}

      {!showIntro && currentScreen === 'project_detail' && (
        <ProjectPage 
          project={selectedProject} 
          projects={projects} 
          currentUsername={user.username}
          user={user}
          onBack={() => setCurrentScreen('main_page')}
          onViewProjectDetails={(nextProject) => setSelectedProject(nextProject)} 
          onOpenMyProjects={() => setCurrentScreen('my_projects')}
        />
      )}

      {!showIntro && currentScreen === 'my_projects' && (
        <MyProjectsPage 
          user={user} 
          projects={projects} 
          onBack={() => setCurrentScreen('main_page')} 
          onDeleteProject={handleApiDeleteProject}
          onViewProjectDetails={(project) => {
            setSelectedProject(project);
            setCurrentScreen('project_detail');
          }}
        />
      )}
    </>
  );
}

export default App;
